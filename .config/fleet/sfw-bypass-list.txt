# Socket Firewall bypass list — fleet-canonical source of truth.
#
# This file is the wheelhouse-tracked master. Every fleet repo gets a
# byte-identical copy at <repo>/.config/sfw-bypass-list.txt via the
# sync-scaffolding cascade, and consumers read their local copy to
# populate SFW_CUSTOM_REGISTRIES:
#
#   - socket-registry  → .github/actions/setup/action.yml grep-reads
#                        this list into SFW_CUSTOM_REGISTRIES in CI.
#   - socket-btm et al → scripts/fleet/install-sfw.mts writes the env to
#                        ~/.socket/_wheelhouse/env.sh so local sfw gets the
#                        same set the CI shared worker uses.
#
# Edit ONLY in socket-wheelhouse/template/.config/sfw-bypass-list.txt
# — downstream copies are regenerated by the cascade and any local
# fork will be clobbered on the next sync.
#
# Format: one `<kind>:<fqdn>` entry per line. Comments + blank lines
# ignored. Kinds accepted: npm, pypi, golang, maven, gem, cargo, nuget,
# block, wrap, bypass. The bundled defaults baked into sfw live at
# SocketDev/firewall:src/lib/registries/default.ts — entries here ship
# *over* a binary release without waiting for a re-publish.
#
# When adding a host, group it with the ecosystem comment that owns it
# and keep within-group ordering stable so cascades produce minimal
# diffs.

# rustup component/toolchain dist CDN (rust-std-wasm32-unknown-unknown
# et al). rustup downloads with rustls, which rejects SFW's MITM cert
# chain outright — `invalid peer certificate: UnknownIssuer` — so a
# `rustup target add` inside a proxied build step fails without this.
# Same shape as the proxy.golang.org entries below. Integrity stays
# enforced by rustup's own per-component SHA256 verification.
# NOTE: deliberate OVERRIDE — the sfw >= 1.13.1 built-in default is
# `wrap` (still MITM'd), which rustls still rejects.
bypass:static.rust-lang.org
bypass:bootstrap.pypa.io
# OrbStack — recommended macOS Docker runtime for Dockerfile-based builds
# (socket-btm glibc/musl node-smol images). The /download endpoint on
# orbstack.dev 307-redirects to the cdn-updates.orbstack.dev .dmg; a
# `brew install --cask orbstack` onboarding step hits both.
bypass:orbstack.dev
bypass:cdn-updates.orbstack.dev

# Go module proxy + toolchain auto-download. proxy.golang.org serves
# modules and 302s to storage.googleapis.com for the
# `go: downloading go1.x.y` flow; sum.golang.org is the checksum DB.
# Without these, `go build` fails with `tls: failed to verify
# certificate: x509: certificate signed by unknown authority` because
# SFW's MITM breaks the cert chain at the redirect target.
# NOTE: deliberate OVERRIDES — sfw >= 1.13.1 ships a built-in `golang`
# scanning kind for both hosts, but the go toolchain does not trust
# the MITM CA, so bypass wins. Remove only after a proxied `go build`
# passes without them.
bypass:proxy.golang.org
bypass:sum.golang.org
bypass:storage.googleapis.com

# Cargo web portal (search/info API). The sparse-index and tarball hosts
# (index.crates.io, static.crates.io) stay PROXIED on purpose: sfw-enterprise
# intercepts at the system level, and its bypass path hands cargo a cert
# chain schannel rejects ("untrusted root" on every registry fetch —
# ultrathink runs 30226404634/30228147887), while the proxied path verifies
# via the CA sfw installs at setup. The sparse-index request storm is tamed
# client-side with CARGO_HTTP_MULTIPLEXING=false instead.
bypass:crates.io
